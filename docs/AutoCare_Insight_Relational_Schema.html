1. Formal Academic Relational Schema Notation (3NF / BCNF)

USERS (user_id, full_name, email, password_hash, role, phone, is_active, created_at, updated_at)

CUSTOMERS (customer_id, full_name, email, phone, address, city, state, zip_code, created_at, updated_at)

VEHICLE_CATALOGUE (catalogue_id, manufacturer, model, model_year, fuel_code, source_dataset, category, fuel, engine_type, engine_size, engine_cylinders, engine_description, transmission_type, drivetrain, alt_economy_city, alt_economy_hwy, alt_economy_combined, conv_economy_city, conv_economy_hwy, conv_economy_combined, electric_range, phev_total_range, num_passengers, heavy_duty_power_system, source_vehicle_id, notes, created_at)

CUSTOMER_VEHICLES (vehicle_id, customer_id, catalogue_id, license_plate, vin, color, purchase_year, current_odometer, fuel_override, notes, is_active, created_at, updated_at)
  • Foreign Key: customer_id → CUSTOMERS(customer_id) [ON DELETE RESTRICT, ON UPDATE CASCADE]
  • Foreign Key: catalogue_id → VEHICLE_CATALOGUE(catalogue_id) [ON DELETE SET NULL, ON UPDATE CASCADE]

SERVICE_TYPES (service_type_id, service_name, description, estimated_duration_mins, base_price, is_active, created_at)

APPOINTMENTS (appointment_id, customer_id, vehicle_id, advisor_id, scheduled_at, status, service_type_id, mileage_at_booking, customer_notes, advisor_notes, created_at, updated_at)
  • Foreign Key: customer_id → CUSTOMERS(customer_id) [ON DELETE RESTRICT]
  • Foreign Key: vehicle_id → CUSTOMER_VEHICLES(vehicle_id) [ON DELETE RESTRICT]
  • Foreign Key: advisor_id → USERS(user_id) [ON DELETE SET NULL]
  • Foreign Key: service_type_id → SERVICE_TYPES(service_type_id) [ON DELETE SET NULL]

SERVICE_JOBS (job_id, appointment_id, mechanic_id, vehicle_id, status, odometer_in, odometer_out, started_at, completed_at, mongo_report_id, internal_notes, created_at, updated_at)
  • Foreign Key: appointment_id → APPOINTMENTS(appointment_id) [ON DELETE SET NULL]
  • Foreign Key: mechanic_id → USERS(user_id) [ON DELETE SET NULL]
  • Foreign Key: vehicle_id → CUSTOMER_VEHICLES(vehicle_id) [ON DELETE RESTRICT]

JOB_SERVICES (job_service_id, job_id, service_type_id, service_price, technician_notes)
  • Foreign Key: job_id → SERVICE_JOBS(job_id) [ON DELETE CASCADE]
  • Foreign Key: service_type_id → SERVICE_TYPES(service_type_id) [ON DELETE RESTRICT]

SPARE_PARTS (part_id, part_number, part_name, description, category, brand, unit_cost, selling_price, qty_in_stock, reorder_level, location_code, is_active, created_at, updated_at)

PART_USAGE (usage_id, job_id, part_id, qty_used, unit_price_charged, used_at)
  • Foreign Key: job_id → SERVICE_JOBS(job_id) [ON DELETE RESTRICT]
  • Foreign Key: part_id → SPARE_PARTS(part_id) [ON DELETE RESTRICT]

INVOICES (invoice_id, job_id, customer_id, subtotal, tax_rate, tax_amount, discount_amount, total_amount, status, due_date, issued_at, created_at, updated_at)
  • Foreign Key (1:1): job_id → SERVICE_JOBS(job_id) [ON DELETE RESTRICT]
  • Foreign Key: customer_id → CUSTOMERS(customer_id) [ON DELETE RESTRICT]

INVOICE_PAYMENTS (payment_id, invoice_id, amount_paid, payment_method, reference_number, paid_at, recorded_by, notes)
  • Foreign Key: invoice_id → INVOICES(invoice_id) [ON DELETE RESTRICT]
  • Foreign Key: recorded_by → USERS(user_id) [ON DELETE SET NULL]

MAINTENANCE_REMINDERS (reminder_id, vehicle_id, service_type_id, reminder_type, next_service_date, next_service_odometer, interval_km, interval_days, is_sent, is_active, created_by, created_at, updated_at)
  • Foreign Key: vehicle_id → CUSTOMER_VEHICLES(vehicle_id) [ON DELETE CASCADE]
  • Foreign Key: service_type_id → SERVICE_TYPES(service_type_id) [ON DELETE SET NULL]
  • Foreign Key: created_by → USERS(user_id) [ON DELETE SET NULL]
